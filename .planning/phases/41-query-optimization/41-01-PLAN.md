<objective>
Move JavaScript-based aggregations to PostgreSQL for timeseries, analytics, and pipeline KPI routes.

Purpose: Eliminate over-fetching and JS loops by computing aggregations at the database level. Currently, routes fetch thousands of orders and aggregate in memory using Map/Set operations. Moving to SQL DATE_TRUNC and AVG/COUNT operations reduces bandwidth, memory, and CPU usage by 10-100x.

Output: Optimized API routes using Prisma $queryRaw for SQL aggregations, plus partial indexes for common filter patterns.
</objective>

```typescript
interface TimeseriesRow {
  date: Date;
  eingang: bigint;
  versendet: bigint;
}

interface AnalyticsRow {
  date: Date;
  count: bigint;
}

// getTimeseries: Replace JS Map aggregation with SQL DATE_TRUNC
export async function getTimeseries(fromDate: string, toDate: string) {
  const result = await prisma.$queryRaw<TimeseriesRow[]>`
    WITH date_series AS (
      SELECT generate_series(
        ${fromDate}::DATE,
        ${toDate}::DATE,
        '1 day'::INTERVAL
      )::DATE as date
    ),
    eingang_counts AS (
      SELECT DATE_TRUNC('day', "createdAt" AT TIME ZONE 'Europe/Berlin')::DATE as date,
             COUNT(*) as count
      FROM "Auftrag"
      WHERE "createdAt" >= ${fromDate}::TIMESTAMPTZ
        AND "createdAt" < (${toDate}::DATE + 1)::TIMESTAMPTZ
      GROUP BY 1
    ),
    versendet_counts AS (
      SELECT DATE_TRUNC('day', COALESCE("versandUpdatedAt", "createdAt") AT TIME ZONE 'Europe/Berlin')::DATE as date,
             COUNT(*) as count
      FROM "Auftrag"
      WHERE "versandStatus" = 'versendet'
        AND COALESCE("versandUpdatedAt", "createdAt") >= ${fromDate}::TIMESTAMPTZ
        AND COALESCE("versandUpdatedAt", "createdAt") < (${toDate}::DATE + 1)::TIMESTAMPTZ
      GROUP BY 1
    )
    SELECT
      ds.date,
      COALESCE(ec.count, 0)::BIGINT as eingang,
      COALESCE(vc.count, 0)::BIGINT as versendet
    FROM date_series ds
    LEFT JOIN eingang_counts ec ON ds.date = ec.date
    LEFT JOIN versendet_counts vc ON ds.date = vc.date
    ORDER BY ds.date
  `;
  // Convert BigInt to number for JSON serialization
  return result.map(row => ({
    date: format(row.date, 'yyyy-MM-dd'),
    eingang: Number(row.eingang),
    versendet: Number(row.versendet),
  }));
}
```

```typescript
interface PeriodKpisRow {
  avg_days_to_ship: number | null;
  on_time_percent: number | null;
  total_shipped: bigint;
}

export async function getPeriodKpis(fromDate: string, toDate: string): Promise<{
  avgDaysToShip: number | null;
  onTimePercent: number;
  totalShipped: number;
}> {
  const result = await prisma.$queryRaw<PeriodKpisRow[]>`
    SELECT
      ROUND(AVG(
        EXTRACT(EPOCH FROM ("versandUpdatedAt" - "createdAt")) / 86400
      )::NUMERIC, 1) as avg_days_to_ship,
      ROUND(
        (COUNT(*) FILTER (WHERE "versandUpdatedAt" <= "liefertermin")::NUMERIC /
         NULLIF(COUNT(*) FILTER (WHERE "liefertermin" IS NOT NULL), 0)) * 100
      , 0) as on_time_percent,
      COUNT(*) as total_shipped
    FROM "Auftrag"
    WHERE "versandStatus" = 'versendet'
      AND (
        ("versandUpdatedAt" >= ${fromDate}::TIMESTAMPTZ AND "versandUpdatedAt" < (${toDate}::DATE + 1)::TIMESTAMPTZ)
        OR "versandUpdatedAt" IS NULL
      )
  `;

  const row = result[0];
  return {
    avgDaysToShip: row?.avg_days_to_ship ?? null,
    onTimePercent: Number(row?.on_time_percent ?? 0),
    totalShipped: Number(row?.total_shipped ?? 0),
  };
}
```

```typescript
/**
 * Ensure partial indexes exist for common query patterns.
 * Safe to call multiple times (IF NOT EXISTS).
 * Run once at startup or via health check.
 */
export async function ensureOptimizationIndexes(): Promise<void> {
  // Index for "not shipped" queries (covers dashboard KPIs, versand list)
  await prisma.$executeRaw`
    CREATE INDEX IF NOT EXISTS idx_auftrag_not_shipped
    ON "Auftrag" ("liefertermin", "istStatus")
    WHERE "versandStatus" != 'versendet'
  `;

  // Index for shipped orders by date (covers timeseries, period KPIs)
  await prisma.$executeRaw`
    CREATE INDEX IF NOT EXISTS idx_auftrag_shipped_date
    ON "Auftrag" ("versandUpdatedAt")
    WHERE "versandStatus" = 'versendet'
  `;

  // Index for problem orders
  await prisma.$executeRaw`
    CREATE INDEX IF NOT EXISTS idx_auftrag_problems
    ON "Auftrag" ("liefertermin")
    WHERE "istStatus" = 'problem' AND "versandStatus" != 'versendet'
  `;

  // Index for orders by creation date (covers eingang counts)
  await prisma.$executeRaw`
    CREATE INDEX IF NOT EXISTS idx_auftrag_created
    ON "Auftrag" ("createdAt")
  `;
}
```
