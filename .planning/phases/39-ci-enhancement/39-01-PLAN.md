          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s

      - Checkout
      - Setup Node.js 20 with npm cache
      - npm ci
      - Cache Playwright browsers
      - Install Playwright (chromium only)
      - Prisma migrate deploy
      - Build with real DB connection
      - Run E2E tests
