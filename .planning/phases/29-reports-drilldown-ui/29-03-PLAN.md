Wire the ReportsOrdersDialog to the Reports page and connect all click handlers.

1. Import ReportsOrdersDialog:
   ```typescript
   import { ReportsOrdersDialog } from '@/components/reports/ReportsOrdersDialog';
   ```

2. Add dialog state:
   ```typescript
   const [dialogState, setDialogState] = useState<{
     isOpen: boolean;
     type: 'problem' | 'oldest' | 'tomorrow' | 'stage';
     stage?: string;
     title: string;
   }>({
     isOpen: false,
     type: 'problem',
     title: '',
   });
   ```

3. Create handlers for different click sources:

   For SnapshotKPIs (onKpiClick):
   ```typescript
   const handleKpiClick = (type: 'problem' | 'oldest' | 'tomorrow') => {
     const titles = {
       problem: 'Problem-Aufträge',
       oldest: 'Älteste Aufträge',
       tomorrow: 'Morgen fällig',
     };
     setDialogState({
       isOpen: true,
       type,
       title: titles[type],
     });
   };
   ```

   For FunnelChart and StageDistributionChart (onStageClick):
   ```typescript
   const handleStageClick = (stage: string) => {
     const stageNames: Record<string, string> = {
       offen: 'Offen',
       in_produktion: 'In Produktion',
       fertig: 'Fertig',
       versandbereit: 'Versandbereit',
       versendet: 'Versendet',
       problem: 'Problem',
     };
     setDialogState({
       isOpen: true,
       type: 'stage',
       stage,
       title: `Aufträge: ${stageNames[stage] || stage}`,
     });
   };
   ```

4. Pass handlers to components:
   ```tsx
   <SnapshotKPIs data={...} loading={...} onKpiClick={handleKpiClick} />
   <FunnelChart data={...} onStageClick={handleStageClick} />
   <StageDistributionChart data={...} total={...} onStageClick={handleStageClick} />
   ```

5. Render dialog at page bottom:
   ```tsx
   <ReportsOrdersDialog
     kpiType={dialogState.type}
     stage={dialogState.stage}
     title={dialogState.title}
     isOpen={dialogState.isOpen}
     onOpenChange={(open) => setDialogState(prev => ({ ...prev, isOpen: open }))}
   />
   ```
